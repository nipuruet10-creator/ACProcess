    const results = [];

    function assert(suite, description, condition, details = "") {
      results.push({
        suite,
        description,
        passed: Boolean(condition),
        details: condition ? "OK" : (details || "Assertion failed")
      });
    }

    async function runTestSuite() {
      try {
        // Setup isolated test instances
      const testMonth = "TEST-2026";
      localStorage.removeItem("walton_pd_test_workbooks");
      localStorage.removeItem("walton_pd_test_breakdown");

      const geminiClient = new GeminiClient();
      const photoManager = new PhotoManager();
      const pptxGenerator = new PPTXGenerator();

      const wbMgr = new MonthWorkbookManager("walton_pd_test_workbooks");
      wbMgr.workbooks[testMonth] = [];
      const breakdown = new AIBreakdownSheet("walton_pd_test_breakdown");
      const syncEngine = new SyncEngine(wbMgr, breakdown, geminiClient, photoManager);

      // ----------------------------------------------------
      // TEST CASE 1 & 2: Add one task -> Confirm one slide
      // ----------------------------------------------------
      const t1 = wbMgr.addTask(testMonth, "Sazzad", "Compressor Foil Cutting System", "YES");
      assert("Test 1", "Add one task generates an immutable Task ID", t1.task_id && t1.task_id.startsWith(`${testMonth}-`), `ID: ${t1.task_id}`);

      let syncRes = await syncEngine.syncMonth(testMonth);
      assert("Test 2", "Confirm exactly one slide appears after sync", syncRes.active_slides === 1 && syncRes.slides.length === 1, `Slides: ${syncRes.active_slides}`);

      // ----------------------------------------------------
      // TEST CASE 3 & 4: Add second task -> Confirm second slide
      // ----------------------------------------------------
      const t2 = wbMgr.addTask(testMonth, "Sazzad", "Smart QR Scanning System", "YES");
      assert("Test 3", "Add second task for same engineer creates distinct Task ID", t2.task_id !== t1.task_id, `t1: ${t1.task_id}, t2: ${t2.task_id}`);

      syncRes = await syncEngine.syncMonth(testMonth);
      assert("Test 4", "Confirm two slides appear (1 Row = 1 Slide Rule)", syncRes.active_slides === 2 && syncRes.slides.length === 2, `Slides: ${syncRes.active_slides}`);

      // ----------------------------------------------------
      // TEST CASE 5 & 6: Change task name -> Confirm same task_id updates
      // ----------------------------------------------------
      const updatedT1 = wbMgr.updateTask(testMonth, t1.task_id, { task_name: "Compressor Foil Cutting Line Automation & Upgrade" });
      assert("Test 5", "Task name updated successfully in workbook", updatedT1.task_name.includes("Automation"));
      assert("Test 6", "Task ID remains strictly unchanged after task name modification", updatedT1.task_id === t1.task_id, `Expected: ${t1.task_id}, Got: ${updatedT1.task_id}`);

      // ----------------------------------------------------
      // TEST CASE 7 & 8: Upload photo later -> Confirm photo appears in slide
      // ----------------------------------------------------
      const samplePhoto = "data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mNk+M9QDwADhgGAWjR9awAAAABJRU5ErkJggg==";
      photoManager.setTaskPhoto(t1.task_id, "before_photo", samplePhoto);
      syncRes = await syncEngine.syncMonth(testMonth);
      const slide1 = syncRes.slides.find(s => s.task_id === t1.task_id);
      assert("Test 7", "Photo attached to Task ID in photo store", photoManager.hasPhoto(t1.task_id));
      assert("Test 8", "Photo appears bound to correct slide in active slides", slide1 && slide1.photo_before === samplePhoto);

      // ----------------------------------------------------
      // TEST CASE 9 & 10: Remove report inclusion -> Confirm slide disappears on sync
      // ----------------------------------------------------
      wbMgr.toggleInclude(testMonth, t2.task_id); // Set t2 to NO
      const t2State = wbMgr.getTasksForMonth(testMonth).find(t => t.task_id === t2.task_id);
      assert("Test 9", "Task inclusion set to NO", t2State.include_in_report === "NO");

      syncRes = await syncEngine.syncMonth(testMonth);
      assert("Test 10", "Excluded task's slide removed from active deck on sync", syncRes.active_slides === 1 && !syncRes.slides.some(s => s.task_id === t2.task_id));

      // ----------------------------------------------------
      // TEST CASE 11 & 12: Add 20 tasks -> Confirm 20 slides
      // ----------------------------------------------------
      const bulkMonth = "BULK-2027";
      wbMgr.workbooks[bulkMonth] = [];
      for (let i = 1; i <= 20; i++) {
        wbMgr.addTask(bulkMonth, "Rafi", `Process Engineering Task #${i} for Tooling Section`, "YES");
      }
      assert("Test 11", "20 tasks added to bulk month", wbMgr.getTasksForMonth(bulkMonth).length === 20);

      const bulkSync = await syncEngine.syncMonth(bulkMonth);
      assert("Test 12", "Confirm exactly 20 individual slides generated for 20 tasks (1 Row = 1 Slide)", bulkSync.active_slides === 20, `Got: ${bulkSync.active_slides}`);

      // ----------------------------------------------------
      // TEST CASE 13: Check editable PPTX generation
      // ----------------------------------------------------
      const testSlideData = SlideLayoutEngine.renderTaskSlide(syncRes.slides[0], 1, 1);
      assert("Test 13", "Editable PPTX generator loaded and supports 16:9 layout", typeof pptxGenerator !== 'undefined' && pptxGenerator.generatePresentation);

      // ----------------------------------------------------
      // TEST CASE 14: Check PDF export structure
      // ----------------------------------------------------
      assert("Test 14", "PDF export engine available with vector 16:9 print styling", typeof PDFReportGenerator !== 'undefined' && PDFReportGenerator.generatePDF);

      // ----------------------------------------------------
      // TEST CASE 16: Repeated sync idempotency
      // ----------------------------------------------------
      const syncRun1 = await syncEngine.syncMonth(testMonth);
      const syncRun2 = await syncEngine.syncMonth(testMonth);
      assert("Test 16", "Running sync multiple times creates 0 duplicate slides (Idempotent)", syncRun1.active_slides === syncRun2.active_slides && syncRun2.added === 0 && syncRun2.updated === 0);

      // ----------------------------------------------------
      // TEST CASE 17: AI Zero Hallucination Guarantee
      // ----------------------------------------------------
      const transformed = geminiClient.templates.localFactualTransform({
        task_name: "New Die Development",
        engineer: "Kamrul"
      });
      const hasFabricatedPercentage = /25%|40%|50%|80%/.test(transformed.ai_description);
      const hasFabricatedCost = /BDT \d+|USD \d+|\$\d+/.test(transformed.ai_description);
      assert("Test 17", "AI does not fabricate unsupported cost savings or percentages", !hasFabricatedPercentage && !hasFabricatedCost, `Desc: "${transformed.ai_description}"`);

      // ----------------------------------------------------
      // ADDITIONAL ARCHITECTURAL ASSERTIONS
      // ----------------------------------------------------
      assert("Architecture", "Lexend is mandated font across SLIDE_THEME", SLIDE_THEME.TYPOGRAPHY.FONT_FAMILY === "Lexend");
      assert("Architecture", "Walton Corporate reference colors configured", SLIDE_THEME.COLORS.NAVY_PRIMARY === "#0B2038" && SLIDE_THEME.COLORS.ORANGE_ACCENT === "#FF6B00");
      assert("Architecture", "Radial progress gauge SVG component renders correctly", RadialGaugeComponent.renderSVG(100, "Completed").includes("100%"));

      // ----------------------------------------------------
      // WALTON EXECUTIVE RED MASTER SLIDE ASSERTIONS (media_1789014742407.jpg)
      // ----------------------------------------------------
      const redTransformed = geminiClient.templates.localFactualTransform({
        task_name: "Compressor Jacket Foil Cutting System Development",
        engineer: "Engr. Sazzad (50463)"
      });
      assert("Walton Red Executive", "Generates balanced split title (Charcoal line 1 + Red line 2)", 
        redTransformed.split_title_1.length > 0 && redTransformed.split_title_2.length > 0, 
        `L1: "${redTransformed.split_title_1}" | L2: "${redTransformed.split_title_2}"`);

      assert("Walton Red Executive", "Generates 4 impact checklist points and 3 metric trend pills", 
        redTransformed.ai_impact.length === 4 && redTransformed.metrics.length === 3 && redTransformed.quote.length > 0,

      const execRedHtml = SlideLayoutEngine.renderTaskSlide({

    function renderResults() {
      const out = document.getElementById('test-output');
      const total = results.length;
      const passed = results.filter(r => r.passed).length;
      const failed = total - passed;

      const summaryText = `TOTAL TESTS: ${total} | PASSED: ${passed} | FAILED: ${failed} | SUCCESS RATE: ${Math.round((passed / total) * 100)}%`;
      console.log(`\n============================================================\n${summaryText}\n============================================================\n`);

      let html = `<div class="summary-box ${failed === 0 ? 'test-pass' : 'test-fail'}">
        ${summaryText}
      </div>`;

      // Group by suite
      const suites = {};
      results.forEach(r => {
        if (!suites[r.suite]) suites[r.suite] = [];
        suites[r.suite].push(r);
      });

      for (const [sName, sResults] of Object.entries(suites)) {
        html += `<div class="suite-card">
          <div class="suite-title">${sName}</div>`;
        sResults.forEach(r => {
          html += `<div class="test-result ${r.passed ? 'test-pass' : 'test-fail'}">
            <span>[${r.passed ? 'PASS' : 'FAIL'}] ${r.description}</span>
            <span>${r.details}</span>
          </div>`;
        });
        html += `</div>`;
      }

      out.innerHTML = html;
    }

    // Run immediately
    runTestSuite();
