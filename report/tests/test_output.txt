    function renderResults() {
      const out = document.getElementById('test-output');
      const suites = {};

      results.forEach(r => {
        if (!suites[r.suite]) suites[r.suite] = [];
        suites[r.suite].push(r);
      });

      let html = '';
      let totalPass = 0;
      let totalFail = 0;

      for (const [suiteName, tests] of Object.entries(suites)) {
        html += `<div class="suite-card"><div class="suite-title">${suiteName} Verification</div>`;
        tests.forEach(t => {
          if (t.passed) totalPass++; else totalFail++;
          html += `<div class="test-result ${t.passed ? 'test-pass' : 'test-fail'}">
            <span>${t.passed ? '✔ PASS' : '✖ FAIL'}: ${t.description}</span>
            <span>${t.details}</span>
          </div>`;
        });
        html += `</div>`;
      }

      const allPassed = totalFail === 0;
      html += `<div id="test-summary" class="summary ${allPassed ? 'test-pass' : 'test-fail'}">
        Summary: ${totalPass} Passed, ${totalFail} Failed. Phase 1 & 2 requirements met: ${allPassed ? 'YES' : 'NO'}.
      </div>`;
      out.innerHTML = html;
    }
